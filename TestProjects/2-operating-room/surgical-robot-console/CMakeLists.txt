# Standalone project that CONSUMES harpia-generated C++ as a black box -- it does
# not depend on the harpia repo, only on a project you generate from this folder
# with run_harpia.sh (pointed at via -DHARPIA_GEN). Modelled on
# examples/consumer/CMakeLists.txt; see that file for the Windows / TLS /
# PostgreSQL branches deliberately left out here for brevity.
cmake_minimum_required(VERSION 3.13)
project(harpia_surgical_robot_console CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT DEFINED HARPIA_GEN)
    message(FATAL_ERROR
        "Set -DHARPIA_GEN=<path to a harpia-generated project> "
        "(the folder run_harpia.sh wrote, containing generated/ and third_party/).")
endif()
# CMake resolves relative paths in file(GLOB) against the source dir, not the
# caller's cwd -- a relative -DHARPIA_GEN silently finds nothing. Require absolute.
if(NOT IS_ABSOLUTE "${HARPIA_GEN}")
    message(FATAL_ERROR
        "-DHARPIA_GEN must be an ABSOLUTE path (got: ${HARPIA_GEN}). "
        "Inside docker/run.sh the repo is at /harpia, so use "
        "-DHARPIA_GEN=/harpia/TestProjects/_gen/<project>.")
endif()
set(GEN "${HARPIA_GEN}/generated/cpp")

find_package(Threads REQUIRED)
find_package(Protobuf REQUIRED)

# Every generated message's protobuf C++ (names are md5-hash-qualified; globbed
# so this file never has to know the hash).
file(GLOB HARPIA_PB_CC CONFIGURE_DEPENDS "${GEN}/protofiles/*.pb.cc")
if(NOT HARPIA_PB_CC)
    message(FATAL_ERROR "No generated protofiles under ${GEN}/protofiles -- "
        "generate a project from this folder first (see README.md).")
endif()
add_library(harpia_msgs STATIC ${HARPIA_PB_CC})
target_include_directories(harpia_msgs PUBLIC "${GEN}")
target_link_libraries(harpia_msgs PUBLIC protobuf::libprotobuf)

add_library(harpia_tinyxml2 STATIC "${HARPIA_GEN}/third_party/tinyxml2/tinyxml2.cpp")
target_include_directories(harpia_tinyxml2 PUBLIC "${HARPIA_GEN}/third_party/tinyxml2")

# Turn every generated message header into one umbrella include, so the
# hand-written sources below never spell out an md5 hash. We include the
# protobuf message headers (protofiles/*.pb.h) rather than the generated
# json/*_json.h convenience adapters: each *_json.h defines a non-overloaded
# harpia::json::is_valid_json(const std::string&), so at most one may appear
# per translation unit (see examples/consumer, which includes exactly one).
# MessageToJsonString from protobuf itself works for every message type.
file(GLOB _pb_h RELATIVE "${GEN}" "${GEN}/protofiles/*.pb.h")
set(_umbrella "${CMAKE_BINARY_DIR}/harpia_generated_includes.h")
file(WRITE "${_umbrella}" "// generated by CMake from the harpia output tree\n#pragma once\n")
foreach(_h ${_pb_h})
    file(APPEND "${_umbrella}" "#include \"${_h}\"\n")
endforeach()

set(HARPIA_INC
    "${CMAKE_BINARY_DIR}"
    "${GEN}"
    "${HARPIA_GEN}/third_party/crow"
    "${HARPIA_GEN}/third_party/asio")
set(HARPIA_LIBS
    harpia_msgs harpia_tinyxml2
    soci_core soci_sqlite3
    Threads::Threads ${CMAKE_DL_LIBS})

add_executable(device_app src/main.cpp)
target_include_directories(device_app PRIVATE ${HARPIA_INC})
target_link_libraries(device_app PRIVATE ${HARPIA_LIBS})
add_executable(human_mock src/human_mock.cpp)
target_include_directories(human_mock PRIVATE ${HARPIA_INC})
target_link_libraries(human_mock PRIVATE ${HARPIA_LIBS})
